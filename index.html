function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Data"); var rows = sheet.getDataRange().getValues(); var headers = rows[0]; var data = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; var obj = {}; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = row[j]; } data.push(obj); } return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function normalizeDate(dateStr) { if (!dateStr) return ""; var cleanStr = String(dateStr).trim(); if (/^\d{4}-\d{2}-\d{2}$/.test(cleanStr)) { var p = cleanStr.split('-'); return p[2] + "-" + p[1] + "-" + p[0]; } return cleanStr; } function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Data"); var requestData = JSON.parse(e.postData.contents); function calculateSOBAndPoint(company) { var comp = (company || "").toLowerCase(); if (comp.includes("agoda") || comp.includes("traveloka") || comp.includes("tiket") || comp.includes("booking") || comp.includes("expedia")) { return { sob: "OTA", point: 1 }; } else if (comp.includes("fit") || comp.includes("walk in") || comp.includes("personal")) { return { sob: "FIT / WIG", point: 1 }; } else { return { sob: "CORP / GOV", point: 2 }; } } // Ambil data yang sudah ada di sheet untuk validasi anti-double var existingRows = sheet.getDataRange().getValues(); var existingPhones = {}; // Lewati baris header (index 0) for (var r = 1; r < existingRows.length; r++) { var phoneCol = String(existingRows[r][2]).replace(/['"]/g, "").trim(); var checkInCol = String(existingRows[r][3]).trim(); var checkOutCol = String(existingRows[r][4]).trim(); if (phoneCol) { // Buat kunci unik berdasarkan No HP + Check-In + Check-Out existingPhones[phoneCol + "_" + checkInCol + "_" + checkOutCol] = true; } } var lastRow = sheet.getLastRow(); var currentId = 0; if (lastRow > 1) { var lastIdVal = sheet.getRange(lastRow, 1).getValue(); currentId = !isNaN(lastIdVal) ? Number(lastIdVal) : lastRow - 1; } var addedCount = 0; for (var i = 0; i < requestData.length; i++) { var item = requestData[i]; var formattedCheckIn = normalizeDate(item.checkIn || item.checkin); var formattedCheckOut = normalizeDate(item.checkOut || item.checkout); var cleanPhone = String(item.phone || "").replace(/['"]/g, "").trim(); // Kunci unik untuk data yang akan dimasukkan var uniqueKey = cleanPhone + "_" + formattedCheckIn + "_" + formattedCheckOut; // Jika kombinasi No HP dan Tanggal Check-In/Out sudah ada, lewati (Cegah Double!) if (existingPhones[uniqueKey]) { continue; } // Jika belum ada, tandai agar tidak double di batch yang sama existingPhones[uniqueKey] = true; currentId++; var calculated = calculateSOBAndPoint(item.company); sheet.appendRow([ currentId, item.name || "", cleanPhone ? "'" + cleanPhone : "", formattedCheckIn, formattedCheckOut, item.company || "", calculated.sob, item.point !== undefined ? item.point : calculated.point, item.redeemed || 0 ]); addedCount++; } // Auto-formatting tabel var totalRows = sheet.getLastRow(); var lastCol = sheet.getLastColumn(); if (totalRows > 0) { sheet.setColumnWidth(1, 50); sheet.setColumnWidth(2, 300); sheet.setColumnWidth(3, 200); sheet.setColumnWidth(4, 150); sheet.setColumnWidth(5, 150); sheet.setColumnWidth(6, 200); sheet.setColumnWidth(7, 100); sheet.setColumnWidth(8, 75); sheet.setColumnWidth(9, 75); var headerRange = sheet.getRange(1, 1, 1, lastCol); headerRange.setFontFamily("Arial").setFontWeight("bold").setFontSize(15).setHorizontalAlignment("center").setVerticalAlignment("middle").setBackground("#1e2124").setFontColor("#ffffff"); sheet.setRowHeight(1, 45); if (totalRows > 1) { var dataRange = sheet.getRange(2, 1, totalRows - 1, lastCol); sheet.setRowHeights(2, totalRows - 1, 45); dataRange.setFontFamily("Times New Roman").setFontSize(11).setVerticalAlignment("middle").setWrap(true); sheet.getRange(2, 1, totalRows - 1, 3).setHorizontalAlignment("left"); sheet.getRange(2, 4, totalRows - 1, 3).setHorizontalAlignment("center"); sheet.getRange(2, 7, totalRows - 1, 3).setHorizontalAlignment("center"); dataRange.setBorder(true, true, true, true, true, true, "#cbd5e1", SpreadsheetApp.BorderStyle.SOLID); } } return ContentService.createTextOutput(JSON.stringify({status: "success", added: addedCount})).setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.toString()})).setMimeType(ContentService.MimeType.JSON); } }
