function openImportModal() { // Setiap modal import dibuka, pastikan textarea bersih (blank) document.getElementById('importTextarea').value = ''; document.getElementById('importModal').classList.remove('hidden'); setTimeout(() => document.getElementById('importModal').classList.remove('opacity-0'), 10); } function toggleImportModal() { document.getElementById('importModal').classList.add('hidden'); } function processExcelImport() { const textareaEl = document.getElementById('importTextarea'); const text = textareaEl.value.trim(); if (!text) { alert("Kotak teks masih kosong. Silakan paste data Excel terlebih dahulu."); return; } let importDataList = []; let countNew = 0; text.split('\n').forEach(line => { let cols = line.split('\t'); if (cols.length < 5) cols = line.split(/[,;|]/); if (cols.length >= 5) { let name = cols[0].trim(); let phone = cols[1].trim(); let checkIn = normalizeDate(cols[2].trim()); let checkOut = normalizeDate(cols[3].trim()); let comp = cols[4].trim(); let sob = comp.toLowerCase().includes('agoda') || comp.toLowerCase().includes('traveloka') || comp.toLowerCase().includes('tiket') || comp.toLowerCase().includes('booking') ? 'OTA' : 'CORP / GOV'; let pt = sob === 'OTA' ? 1 : 2; importDataList.push({ name, phone, checkIn, checkOut, company: comp, point: pt, redeemed: 0 }); let existing = guestDatabase.find(g => g.phone === phone); if (existing) { existing.point += pt; existing.isRedeemedLocked = false; existing.history.push({ checkIn, checkOut, company: comp, point: pt }); } else { guestDatabase.push({ id: guestDatabase.length + 1, name: name, phone: phone, company: comp, sob: sob, point: pt, redeemed: 0, isRedeemedLocked: false, history: [{ checkIn, checkOut, company: comp, point: pt }] }); } countNew++; } }); if (countNew > 0) { if (SCRIPT_URL) { fetch(SCRIPT_URL, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(importDataList) }).catch(err => console.log('Sync to sheets error:', err)); } // Kosongkan textarea agar bersih untuk import selanjutnya textareaEl.value = ''; filterData(); updateAllTimeStats(); toggleImportModal(); alert(`Berhasil memproses dan mengimpor ${countNew} data tamu!`); } else { alert("Format paste excel tidak valid. Pastikan kolom: Nama | No HP | Check-In | Check-Out | Company."); } }
