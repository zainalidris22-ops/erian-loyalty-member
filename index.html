function filterData() { const search = document.getElementById('searchInput').value.toLowerCase().trim(); const sobVal = document.getElementById('sobFilter').value; filteredData = guestDatabase.filter(item => { // Pencarian global berdasarkan nama atau no HP tanpa batasan bulan const matchSearch = !search || item.name.toLowerCase().includes(search) || item.phone.toLowerCase().includes(search); const matchSOB = sobVal === 'ALL' || item.sob === sobVal; // Jika sedang mengetik di kolom search, abaikan filter bulan agar pencarian menyeluruh if (search) { return matchSearch && matchSOB; } // Jika tidak sedang search, jalankan filter berdasarkan bulan yang dipilih let matchMonth = true; if (item.history) { matchMonth = item.history.some(h => { let parts = h.checkIn.split('-'); if (parts.length === 3) { return (`${parts[2]}-${parts[1]}` === `${currentSelectedYear}-${currentSelectedMonth}`); } return false; }); } return matchSearch && matchSOB && matchMonth; }); document.getElementById('monthlyGuestsCount').innerText = filteredData.length; document.getElementById('monthlyPointsCount').innerText = filteredData.reduce((acc, curr) => acc + (curr.point || 0), 0) + " Pts"; document.getElementById('monthlyRedeemCount').innerText = filteredData.reduce((acc, curr) => acc + (curr.redeemed || 0), 0) + " Pts"; currentPage = 1; renderTable(); }
